# Soviet industry relocation (SOV_industry_relocation_* missions in common/decisions/SOV.txt)
# Run in STATE scope once per mission tick. Moves 4 factories per tick, any mix of military and civilian,
# and rebuilds every moved factory in Siberia as a military factory.
# A type is only taken while the state has more than 2 of it (taking 4 needs more than 4), so states are never fully stripped.
# All checks use the counts at the start of the tick, so it never removes more than the state has.
IHE_SOV_relocate_state_factories = {
	if = {
		limit = { is_owned_and_controlled_by = SOV }
		if = {
			limit = {
				arms_factory > 2
				industrial_complex > 2
			}
			remove_building = { type = arms_factory level = 2 }
			remove_building = { type = industrial_complex level = 2 }
			IHE_SOV_build_relocated_factories = yes
			IHE_SOV_build_relocated_factories = yes
		}
		else_if = {
			limit = { arms_factory > 4 }
			remove_building = { type = arms_factory level = 4 }
			IHE_SOV_build_relocated_factories = yes
			IHE_SOV_build_relocated_factories = yes
		}
		else_if = {
			limit = { industrial_complex > 4 }
			remove_building = { type = industrial_complex level = 4 }
			IHE_SOV_build_relocated_factories = yes
			IHE_SOV_build_relocated_factories = yes
		}
		else_if = {
			limit = { arms_factory > 2 }
			remove_building = { type = arms_factory level = 2 }
			IHE_SOV_build_relocated_factories = yes
		}
		else_if = {
			limit = { industrial_complex > 2 }
			remove_building = { type = industrial_complex level = 2 }
			IHE_SOV_build_relocated_factories = yes
		}
	}
}

# Builds 2 military factories instantly in a random Siberian state (flag set in on_startup, 05_on_actions.txt)
IHE_SOV_build_relocated_factories = {
	random_state = {
		limit = {
			has_state_flag = is_literally_siberia
			free_building_slots = {
				building = arms_factory
				size > 2
				include_locked = yes
			}
		}
		add_extra_state_shared_building_slots = 2
		add_building_construction = {
			type = arms_factory
			level = 2
			instant_build = yes
		}
	}
}

# Rear Area mission (SOV_industry_relocation_rear): run in STATE scope once per tick.
# Converts up to 4 civilian factories into military factories in the same state, never taking the state below 5 civilian
# factories (7 civs -> converts 2, builds 2 mils). Nothing is moved to Siberia and existing military factories are untouched.
IHE_SOV_convert_state_civs = {
	if = {
		limit = { is_owned_and_controlled_by = SOV }
		if = {
			limit = { industrial_complex > 8 }
			remove_building = { type = industrial_complex level = 4 }
			add_building_construction = { type = arms_factory level = 4 instant_build = yes }
		}
		else_if = {
			limit = { industrial_complex > 7 }
			remove_building = { type = industrial_complex level = 3 }
			add_building_construction = { type = arms_factory level = 3 instant_build = yes }
		}
		else_if = {
			limit = { industrial_complex > 6 }
			remove_building = { type = industrial_complex level = 2 }
			add_building_construction = { type = arms_factory level = 2 instant_build = yes }
		}
		else_if = {
			limit = { industrial_complex > 5 }
			remove_building = { type = industrial_complex level = 1 }
			add_building_construction = { type = arms_factory level = 1 instant_build = yes }
		}
	}
}
