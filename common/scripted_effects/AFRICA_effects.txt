###### AFRICA WAR

# global.AFRICA_score = zones held by the Allied side minus zones held by the Axis side (-10..10)
# Every 2 points of score is one level, so taking a zone from the enemy moves one step; all 10 zones = max level
# Allied level = 6 + score/2, Axis level = 6 - score/2 (rounded toward 0)

# Recount who holds the Africa zones and re-apply the levels. Safe to call from any scope.
AFRICA_recalculate = {
	set_variable = { global.AFRICA_score = 0 }
	every_state = {
		limit = { AFRICA_is_zone = yes }
		if = {
			limit = { controller = { AFRICA_is_allied_side = yes } }
			add_to_variable = { global.AFRICA_score = 1 }
		}
		else_if = {
			limit = { controller = { AFRICA_is_axis_side = yes } }
			subtract_from_variable = { global.AFRICA_score = 1 }
		}
	}

	# only swap ideas when the score actually moved (unset AFRICA_score_applied = 0 = stalemate start)
	if = {
		limit = {
			NOT = { check_variable = { var = global.AFRICA_score value = global.AFRICA_score_applied compare = equals } }
		}
		set_variable = { global.AFRICA_score_applied = global.AFRICA_score }
		every_country = {
			limit = { AFRICA_has_allied_idea = yes }
			AFRICA_apply_allied_level = yes
		}
		every_country = {
			limit = { AFRICA_has_axis_idea = yes }
			AFRICA_apply_axis_level = yes
		}
		AFRICA_update_german_armour = yes
		AFRICA_update_western_defences = yes
	}
}

# Strength 0..5 from Britain's current AFRICA_ALLIES level: ALLIES_6+ = 0, ALLIES_1 = 5
# (to flip the direction: ALLIES_7 = 1 ... ALLIES_11 = 5)
AFRICA_update_western_defences = {
	set_variable = { global.AFRICA_west_strength = 0 }
	if = {
		limit = { country_exists = ENG }
		ENG = {
			if = {
				limit = { has_idea = AFRICA_ALLIES_1 }
				set_variable = { global.AFRICA_west_strength = 5 }
			}
			else_if = {
				limit = { has_idea = AFRICA_ALLIES_2 }
				set_variable = { global.AFRICA_west_strength = 4 }
			}
			else_if = {
				limit = { has_idea = AFRICA_ALLIES_3 }
				set_variable = { global.AFRICA_west_strength = 3 }
			}
			else_if = {
				limit = { has_idea = AFRICA_ALLIES_4 }
				set_variable = { global.AFRICA_west_strength = 2 }
			}
			else_if = {
				limit = { has_idea = AFRICA_ALLIES_5 }
				set_variable = { global.AFRICA_west_strength = 1 }
			}
		}
	}

	# Same mechanic as AXIS_Africa_modifier: fixed-value modifiers, one copy per Allied country
	# (scope = FRA/ENG/USA/CAN/SAF/BRA) so only that country's divisions get the attack penalty.
	# Levels 1-5 = strength; per level -10% attack, -1.5% controller attrition, +1.5% enemy attrition.
	every_state = {
		limit = { AFRICA_is_west_defence_state = yes }
		AFRICA_remove_western_defences = yes
		if = {
			limit = {
				check_variable = { global.AFRICA_west_strength > 0 }
				date < 1942.1.1 # the Fortified Western Front modifiers expire in 1942
				controller = { AFRICA_is_west_defender = yes }
			}
			if = {
				limit = { check_variable = { global.AFRICA_west_strength > 4 } }
				AFRICA_add_western_defences_5 = yes
			}
			else_if = {
				limit = { check_variable = { global.AFRICA_west_strength > 3 } }
				AFRICA_add_western_defences_4 = yes
			}
			else_if = {
				limit = { check_variable = { global.AFRICA_west_strength > 2 } }
				AFRICA_add_western_defences_3 = yes
			}
			else_if = {
				limit = { check_variable = { global.AFRICA_west_strength > 1 } }
				AFRICA_add_western_defences_2 = yes
			}
			else = {
				AFRICA_add_western_defences_1 = yes
			}
		}
	}
}

# Atlantic Wall: while the Axis holds AXIS_11 (score -10, all zones), every 120 days the
# AFRICA_dday_forts mission adds 1 coastal fort level to every coastal province of the D-Day states
# (the French Atlantic coast, Benelux and the German North Sea coast). Coastal forts cap at 10, so the counter stops at 10.
AFRICA_add_dday_forts = {
	if = {
		limit = {
			check_variable = { global.AFRICA_dday_forts < 10 }
			date < 1942.1.1
		}
		every_state = {
			limit = {
				OR = {
					state = 19
					state = 23
					state = 30
					state = 14
					state = 15
					state = 785
					state = 29
					state = 6
					state = 7
					state = 35
					state = 36
					state = 56
					state = 58
					state = 59
				}
			}
			add_building_construction = {
				type = coastal_bunker
				level = 1
				instant_build = yes
				province = {
					all_provinces = yes
					limit_to_coastal = yes
				}
			}
		}
		add_to_variable = { global.AFRICA_dday_forts = 1 }
	}
}
